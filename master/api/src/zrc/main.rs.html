74/// Decorate the value at `path` in `config.toml` with a leading `# {comment}`
75/// line, preserving any non-comment whitespace. Mirrors the gateway's
76/// `apply_comments`. Best-effort — silently bails on parse errors so a
77/// successful set isn't downgraded to a failure for a metadata problem.
78async fn apply_comment_inline(
79    config_path: &std::path::Path,
80    path: &str,
81    comment: &str,
82) -> Result<()> {
83    zeroclaw_config::comment_writer::apply_comments(
84        config_path,
85        &[(path.to_string(), comment.to_string())],
86    )
87    .await
88    .context("failed to write comment annotation")
89}

99fn json_value_to_setprop_string(
100    value: &serde_json::Value,
101    config: &Config,
102    path: &str,
103) -> Result<String> {
104    let kind = config_patch_prop_kind(config, path);
105    zeroclaw_config::typed_value::coerce_for_set_prop(value, kind).map_err(|e| {
106        ::zeroclaw_log::record!(
107            WARN,
108            ::zeroclaw_log::Event::new(module_path!(), ::zeroclaw_log::Action::Reject)
109                .with_outcome(::zeroclaw_log::EventOutcome::Failure)
110                .with_attrs(::serde_json::json!({"path": path, "error": e.message.clone()})),
111            "config patch coercion rejected JSON value"
112        );
113        anyhow::Error::msg(e.message)
114    })
115}

117fn config_patch_map_prop_error(err: anyhow::Error, path: &str, op_index: usize) -> ConfigApiError {
118    let msg = err.to_string();
119    if msg.starts_with("Unknown property") {
120        ConfigApiError::path_not_found(path).with_op_index(op_index)
121    } else {
122        ConfigApiError::from_validation(err)
123            .with_path(path)
124            .with_op_index(op_index)
125    }
126}

133fn config_patch_json_value_type_error(
134    message: impl Into<String>,
135    path: Option<String>,
136    op_index: Option<usize>,
137) -> ConfigApiError {
138    let mut err = ConfigApiError::new(ConfigApiCode::ValueTypeMismatch, message.into());
139    if let Some(path) = path {
140        err = err.with_path(path);
141    }
142    if let Some(op_index) = op_index {
143        err = err.with_op_index(op_index);
144    }
145    err
146}

167fn print_no_command_help(cmd: clap::Command) -> Result<()> {
168    #[cfg(feature = "agent-runtime")]
169    {
170        println!(
171            "{}",
172            crate::i18n::get_cli_string("cli-no-command-provided")
173                .as_deref()
174                .unwrap_or("No command provided.")
175        );
176        println!(
177            "{}",
178            crate::i18n::get_cli_string("cli-try-quickstart")
179                .as_deref()
180                .unwrap_or("Try `zeroclaw quickstart` to create your first agent.")
181        );
182    }
183    #[cfg(not(feature = "agent-runtime"))]
184    {
185        println!("{}", t("cli-no-command", "No command provided."));
186        println!(
187            "{}",
188            t(
189                "cli-try-quickstart",
190                "Try `zeroclaw quickstart` to create your first agent."
191            )
192        );
193    }
194    println!();
195
196    let mut cmd = cmd;
197    cmd.print_help()?;
198    println!();
199
200    #[cfg(windows)]
201    pause_after_no_command_help();
202
203    Ok(())
204}

1044/// `zeroclaw quickstart` CLI entry — checklist UX, not a wizard.
1045///
1046/// Mirrors the TUI Quickstart pane's structure: a single screen
1047/// listing all six selectors with `[ ]` / `[✓]` status and a one-line
1048/// summary, the user picks which selector to fill (any order), each
1049/// selector opens its own picker / field-form / channel-list sub-flow,
1050/// and `c` creates the agent once every selector is `[✓]`. There are
1051/// no pre-checked defaults anywhere — every selector starts `[ ]` and
1052/// is only satisfied by an explicit user choice (either a "Use
1053/// existing" pick of an already-configured alias, or a fully-filled
1054/// "Create new" entry).
1055///
1056/// All option lists, field shapes, presets, and the apply path come
1057/// directly from `zeroclaw_runtime::quickstart` — the same module the
1058/// gateway and TUI surfaces consume. No RPC, no daemon: the CLI is
1059/// compiled in-process with `zeroclaw-runtime` and calls
1060/// `snapshot_state` / `field_shape` / `apply_with_surface` as plain
1061/// functions.
1062///
1063/// Flag pre-fills (`--model-provider`, `--model`, `--api-key`,
1064/// `--agent`) silently seed the relevant selector's value and mark it
1065/// `[✓]` if the seed is enough to satisfy the selector; the user can
1066/// still open that selector and overwrite it.
1067#[cfg(feature = "agent-runtime")]
1068async fn run_quickstart_cli(
1069    model_provider: Option<String>,
1070    model: Option<String>,
1071    api_key: Option<String>,
1072    agent: Option<String>,
1073) -> anyhow::Result<()> {
1074    use dialoguer::{Confirm, Editor, FuzzySelect, Input, Password};
1075    use zeroclaw_config::presets::{
1076        AgentIdentity, BuilderSubmission, ChannelQuickStart, MemoryChoice, ModelProviderChoice,
1077        RISK_PRESETS, SelectorChoice,
1078    };
1079    use zeroclaw_runtime::quickstart::{
1080        FieldSection, QuickstartTypeOption, Surface, apply_with_surface, field_shape,
1081        snapshot_state,
1082    };
1083
1084    // ── Form state ──────────────────────────────────────────────
1085    //
1086    // Every field is `Option<…>` and starts `None`. A selector is
1087    // `[✓]` iff its constituent fields are all `Some(_)` and
1088    // non-empty. The form is mutated by the selector sub-flows and
1089    // read by the main checklist render loop.
1090    #[derive(Default)]
1091    struct Form {
1092        provider: Option<ProviderChoice>,
1093        risk: Option<PresetChoice>,
1094        memory: Option<MemoryChoice>,
1095        channels: Vec<ChannelChoice>,
1096        // Tracks whether the user explicitly visited Channels and
1097        // confirmed "no channels". An empty `channels` Vec with
1098        // `channels_visited == false` is *not* satisfied — the
1099        // selector still shows `[ ]`.
1100        channels_visited: bool,
1101        peer_groups: Vec<zeroclaw_config::presets::QuickstartPeerGroup>,
1102        // Mirrors `channels_visited`: peer groups are optional, so an
1103        // empty `peer_groups` Vec only counts as satisfied once the
1104        // user has actually opened the selector and left it. Until
1105        // then the row stays `[ ]` rather than a pre-checked default.
1106        peer_groups_visited: bool,
1107        agent: Option<AgentChoice>,
1108    }
1109    enum ProviderChoice {
1110        Fresh {
1111            kind: String,
1112            display_name: String,
1113            alias: String,
1114            model: String,
1115            /// Round-trip of every non-`model` descriptor value the
1116            /// daemon's `field_shape()` emitted, keyed by descriptor
1117            /// key. The CLI doesn't know what these mean — the daemon
1118            /// authored them and consumes them on the way back.
1119            fields: std::collections::HashMap<String, String>,
1120        },
1121        Existing {
1122            alias_ref: String,
1123        },
1124    }
1125    enum PresetChoice {
1126        Fresh(&'static str),
1127        Existing(String),
1128    }
1129    enum ChannelChoice {
1130        Fresh {
1131            kind: String,
1132            display_name: String,
1133            alias: String,
1134            extras: std::collections::BTreeMap<String, String>,
1135        },
1136        Existing {
1137            alias_ref: String,
1138        },
1139    }
1140    struct AgentChoice {
1141        name: String,
1142        system_prompt: String,
1143        personality_files: Vec<zeroclaw_config::presets::QuickstartPersonalityFile>,
1144    }
1145
1146    impl Form {
1147        fn provider_done(&self) -> bool {
1148            self.provider.is_some()
1149        }
1150        fn risk_done(&self) -> bool {
1151            self.risk.is_some()
1152        }
1153        fn memory_done(&self) -> bool {
1154            self.memory.is_some()
1155        }
1156        fn channels_done(&self) -> bool {
1157            self.channels_visited
1158        }
1159        fn peer_groups_done(&self) -> bool {
1160            self.peer_groups_visited
1161        }
1162        fn agent_done(&self) -> bool {
1163            self.agent
1164                .as_ref()
1165                .is_some_and(|a| !a.name.trim().is_empty())
1166        }
1167        fn all_done(&self) -> bool {
1168            self.provider_done()
1169                && self.risk_done()
1170                && self.memory_done()
1171                && self.channels_done()
1172                && self.agent_done()
1173        }
1174    }
1175
1176    // ── Load config + canonical registries ──────────────────────
1177    let _dirs = crate::config::schema::resolve_runtime_dirs().await?;
1178    let mut cfg = Box::pin(crate::config::schema::Config::load_or_init()).await?;
1179    let state = snapshot_state(&cfg);
1180    let providers: &[QuickstartTypeOption] = &state.model_provider_types;
1181    let channel_types: &[QuickstartTypeOption] = &state.channel_types;
1182    if providers.is_empty() {
1183        anyhow::bail!(
1184            "Quickstart could not enumerate model providers — \
1185             zeroclaw_providers::list_model_providers() returned no entries."
1186        );
1187    }
1188
1189    let mut form = Form::default();
1190
1191    // ── Seed from flags (silent — no UI hit) ────────────────────
1192    //
1193    // Flag-seeded values are recorded into the form so the
1194    // selector renders `[✓]` immediately, but only when the seed
1195    // is enough to satisfy the selector on its own. A bare
1196    // `--model-provider anthropic` without `--model` cannot
1197    // produce a complete `ProviderChoice::Fresh`, so it is
1198    // discarded rather than left half-built — the user opens the
1199    // selector and starts fresh.
1200    if let (Some(mp), Some(m)) = (model_provider.as_deref(), model.as_deref())
1201        && let Some(found) = providers.iter().find(|p| p.kind.eq_ignore_ascii_case(mp))
1202    {
1203        let needs_key = !found.local && api_key.is_none();
1204        if !needs_key {
1205            let mut fields: std::collections::HashMap<String, String> =
1206                std::collections::HashMap::new();
1207            if let Some(key) = api_key.as_deref().filter(|s| !s.is_empty()) {
1208                // Submission field keys are snake_case (`api_key`) — the apply
1209                // path round-trips them verbatim into `set_prop_persistent`,
1210                // which rejects kebab-case with "Unknown property".
1211                fields.insert("api_key".to_string(), key.to_string());
1212            }
1213            form.provider = Some(ProviderChoice::Fresh {
1214                kind: found.kind.clone(),
1215                display_name: found.display_name.clone(),
1216                alias: "default".to_string(),
1217                model: m.to_string(),
1218                fields,
1219            });
1220        }
1221    }
1222    if let Some(a) = agent.as_deref() {
1223        let trimmed = a.trim();
1224        if !trimmed.is_empty() {
1225            form.agent = Some(AgentChoice {
1226                name: trimmed.to_string(),
1227                system_prompt: String::new(),
1228                personality_files: Vec::new(),
1229            });
1230        }
1231    }
1232
1233    // ── Main checklist loop ─────────────────────────────────────
1234    #[derive(Clone, Copy)]
1235    enum Action {
1236        Provider,
1237        Risk,
1238        Memory,
1239        Channels,
1240        PeerGroups,
1241        Agent,
1242        Create,
1243        Quit,
1244    }
1245
1246    println!();
1247    println!(
1248        "{}",
1249        t(
1250            "cli-quickstart-title",
1251            "Quickstart — create one working agent end-to-end."
1252        )
1253    );
1254    println!();
1255
1256    loop {
1257        // Render selector list with current status / summary.
1258        let glyph = |ok: bool| if ok { "[✓]" } else { "[ ]" };
1259        let provider_summary = match &form.provider {
1260            None => "not yet chosen".to_string(),
1261            Some(ProviderChoice::Fresh {
1262                display_name,
1263                alias,
1264                model,
1265                ..
1266            }) => format!("{display_name} (alias: {alias}, model: {model})"),
1267            Some(ProviderChoice::Existing { alias_ref }) => {
1268                format!("use existing {alias_ref}")
1269            }
1270        };
1271        let preset_summary = |p: &Option<PresetChoice>| -> String {
1272            match p {
1273                None => "not yet chosen".to_string(),
1274                Some(PresetChoice::Fresh(name)) => format!("preset: {name}"),
1275                Some(PresetChoice::Existing(a)) => format!("use existing {a}"),
1276            }
1277        };
1278        let memory_summary = match &form.memory {
1279            None => "not yet chosen".to_string(),
1280            Some(kind) => serde_json::to_value(kind)
1281                .ok()
1282                .and_then(|v| v.as_str().map(str::to_string))
1283                .unwrap_or_else(|| format!("{kind:?}").to_lowercase()),
1284        };
1285        let channels_summary = if !form.channels_visited {
1286            "not yet visited".to_string()
1287        } else if form.channels.is_empty() {
1288            "none (chat via `zeroclaw agent` only)".to_string()
1289        } else {
1290            form.channels
1291                .iter()
1292                .map(|c| match c {
1293                    ChannelChoice::Fresh { kind, alias, .. } => format!("{kind}.{alias}"),
1294                    ChannelChoice::Existing { alias_ref } => alias_ref.clone(),
1295                })
1296                .collect::<Vec<_>>()
1297                .join(", ")
1298        };
1299        let agent_summary = match &form.agent {
1300            None => "not yet named".to_string(),
1301            Some(a) => format!(
1302                "alias: {}, system prompt: {} chars, {} personality file(s)",
1303                a.name,
1304                a.system_prompt.len(),
1305                a.personality_files.len(),
1306            ),
1307        };
1308        let peer_groups_summary = if form.peer_groups.is_empty() {
1309            "none — channels accept no peers".to_string()
1310        } else {
1311            form.peer_groups
1312                .iter()
1313                .map(|pg| format!("{} → {}", pg.channel, pg.name))
1314                .collect::<Vec<_>>()
1315                .join(", ")
1316        };
1317
1318        let mut labels: Vec<String> = vec![
1319            format!(
1320                "{} Model provider     — {provider_summary}",
1321                glyph(form.provider_done())
1322            ),

2213#[cfg(feature = "agent-runtime")]
2214fn prompt_for_field(
2215    desc: &zeroclaw_runtime::quickstart::FieldDescriptor,
2216    seed: Option<&str>,
2217) -> anyhow::Result<Option<String>> {
2218    use dialoguer::{FuzzySelect, Input, Password};
2219    use zeroclaw_config::traits::PropKind;
2220    if !desc.help.is_empty() {
2221        println!("  {}", desc.help);
2222    }
2223    let prompt = desc.label.clone();
2224    if desc.is_secret {
2225        // dialoguer 0.12 has no Esc-cancellable Password — only Ctrl+C
2226        // (returns `ErrorKind::Interrupted` wrapped in `dialoguer::Error::IO`).
2227        // Map that to `Ok(None)` so the caller treats it as "user backed
2228        // out" instead of bubbling a confusing IO-error message.
2229        match Password::new()
2230            .with_prompt(prompt.clone())
2231            .allow_empty_password(true)
2232            .interact()
2233        {
2234            Ok(pw) => return Ok(Some(pw)),
2235            Err(e) => {
2236                let io: std::io::Error = e.into();
2237                if io.kind() == std::io::ErrorKind::Interrupted {
2238                    return Ok(None);
2239                }
2240                return Err(io.into());
2241            }
2242        }
2243    }
2244    if let (PropKind::Enum, Some(variants)) = (&desc.kind, &desc.enum_variants) {
2245        let Some(i) = FuzzySelect::new()
2246            .with_prompt(prompt)
2247            .items(variants)
2248            .default(0)
2249            .max_length(variants.len().max(1))
2250            .interact_opt()?
2251        else {
2252            return Ok(None);
2253        };
2254        return Ok(Some(variants[i].clone()));
2255    }
2256    let mut input = Input::<String>::new()
2257        .with_prompt(prompt)
2258        .allow_empty(!desc.required);
2259    if let Some(s) = seed {
2260        input = input.default(s.to_string());
2261    } else if let Some(d) = desc.default.as_deref()
2262        && !d.is_empty()
2263        && d != zeroclaw_config::traits::UNSET_DISPLAY
2264    {
2265        // `<unset>` is a display placeholder for an unset Option, not a
2266        // real default. Seeding it pre-fills the prompt so a bare Enter
2267        // submits `<unset>`, which the daemon then validates against the
2268        // field's true type (e.g. a bool) and rejects.
2269        input = input.default(d.to_string());
2270    }
2271    // Same Ctrl+C-as-cancel mapping as the Password branch above.
2272    match input.interact_text() {
2273        Ok(v) => Ok(Some(v)),
2274        Err(e) => {
2275            let io: std::io::Error = e.into();
2276            if io.kind() == std::io::ErrorKind::Interrupted {
2277                Ok(None)
2278            } else {
2279                Err(io.into())
2280            }
2281        }
2282    }
2283}

2285/// Pick a preset selector — used by both Risk and Runtime since
2286/// their UX is identical (a fixed list of preset rows + the same
2287/// "use existing alias" dual-mode). Returns `None` when the user
2288/// cancels. Returns `Some(Ok(preset_name))` for a fresh preset
2289/// pick or `Some(Err(existing_alias))` for a reuse pick so the
2290/// caller can map into the right `SelectorChoice` variant.
2291#[cfg(feature = "agent-runtime")]
2292fn pick_preset(
2293    prompt: &str,
2294    presets: Vec<(&'static str, &'static str, &'static str)>,
2295    existing: &[String],
2296) -> anyhow::Result<Option<Result<&'static str, String>>> {
2297    use dialoguer::FuzzySelect;
2298    let mut mode_labels: Vec<String> = Vec::new();
2299    let mut mode_kinds: Vec<&str> = Vec::new();
2300    if !existing.is_empty() {
2301        mode_labels.push("Use existing".to_string());
2302        mode_kinds.push("existing");
2303    }
2304    mode_labels.push("Pick a preset".to_string());
2305    mode_kinds.push("preset");
2306    let mode = if mode_labels.len() == 1 {
2307        Some(0)
2308    } else {
2309        FuzzySelect::new()
2310            .with_prompt(prompt)
2311            .items(&mode_labels)
2312            .default(0)
2313            .max_length(mode_labels.len())
2314            .interact_opt()?
2315    };
2316    let Some(mi) = mode else { return Ok(None) };
2317    if mode_kinds[mi] == "existing" {
2318        let Some(i) = FuzzySelect::new()
2319            .with_prompt(format!("Pick an existing {prompt}"))
2320            .items(existing)
2321            .default(0)
2322            .max_length(existing.len().max(1))
2323            .interact_opt()?
2324        else {
2325            return Ok(None);
2326        };
2327        return Ok(Some(Err(existing[i].clone())));
2328    }
2329    let labels: Vec<String> = presets
2330        .iter()
2331        .map(|(_, label, help)| format!("{label}  —  {help}"))
2332        .collect();
2333    let Some(i) = FuzzySelect::new()
2334        .with_prompt(format!("Pick a {prompt} preset"))
2335        .items(&labels)
2336        .default(0)
2337        .max_length(labels.len().max(1))
2338        .interact_opt()?
2339    else {
2340        return Ok(None);
2341    };
2342    Ok(Some(Ok(presets[i].0)))
2343}

2710/// Validate a locale code against the embedded `locales.toml` registry (the
2711/// build's known locales, available in-memory — no file read, no network), so a
2712/// fetch can never be coerced to a path/host outside the known set. Also
2713/// enforces a strict syntactic allowlist as a belt-and-suspenders guard against
2714/// path traversal.
2715#[cfg(feature = "agent-runtime")]
2716fn validated_locale(locale: &str) -> Result<String> {
2717    let ok_shape = !locale.is_empty()
2718        && locale.len() <= 16
2719        && locale
2720            .chars()
2721            .all(|c| c.is_ascii_alphanumeric() || c == '-');
2722    if !ok_shape {
2723        bail!("invalid locale code '{locale}'");
2724    }
2725    let known = zeroclaw_runtime::i18n::available_locales();
2726    if !known.iter().any(|o| o.code == locale) {
2727        let codes: Vec<&str> = known.iter().map(|o| o.code.as_str()).collect();
2728        bail!(
2729            "locale '{locale}' is not in the locales.toml registry; known: {}",
2730            codes.join(", ")
2731        );
2732    }
2733    Ok(locale.to_string())
2734}

2736/// Fetch translated FTL catalogues for `locale` from upstream and install them
2737/// under `<config-dir>/data/ftl/<locale>/`. `catalog` is an optional
2738/// comma-separated subset of {cli, tools, zerocode}; `None` fetches all.
2739///
2740/// Security: `locale` is validated against the upstream `locales.toml` registry
2741/// and a strict syntactic allowlist; the destination path is built from
2742/// `ftl_locale_dir` and canonicalized to confirm it stays under the data dir,
2743/// so neither the locale nor catalog can drive a write outside the FTL store.
2744#[cfg(feature = "agent-runtime")]
2745async fn fetch_locales(locale: &str, catalog: Option<&str>) -> Result<()> {
2746    let locale = validated_locale(locale)?;
2747
2748    let selected: Vec<&(&str, &str, &str)> = match catalog {
2749        None => zeroclaw_config::schema::FTL_CATALOGS.iter().collect(),
2750        Some(list) => {
2751            let names: Vec<&str> = list
2752                .split(',')
2753                .map(str::trim)
2754                .filter(|s| !s.is_empty())
2755                .collect();
2756            let mut out = Vec::new();
2757            for name in &names {
2758                match zeroclaw_config::schema::FTL_CATALOGS
2759                    .iter()
2760                    .find(|(n, _, _)| n == name)
2761                {
2762                    Some(entry) => out.push(entry),
2763                    None => {
2764                        let valid = zeroclaw_config::schema::FTL_CATALOGS
2765                            .iter()
2766                            .map(|(n, _, _)| *n)
2767                            .collect::<Vec<_>>()
2768                            .join(", ");
2769                        bail!("unknown catalog '{name}'; valid: {valid}");
2770                    }
2771                }
2772            }
2773            out
2774        }
2775    };
2776
2777    let dest = zeroclaw_config::schema::ftl_locale_dir(&locale)?;
2778    std::fs::create_dir_all(&dest).with_context(|| format!("creating {}", dest.display()))?;
2779    // Confinement check: the resolved dest must live under the data-dir FTL root.
2780    let ftl_root = dest
2781        .parent()
2782        .map(Path::to_path_buf)
2783        .unwrap_or_else(|| dest.clone());

2785    let canon_root = std::fs::canonicalize(&ftl_root).unwrap_or(ftl_root);
2786    if !canon_dest.starts_with(&canon_root) {
2787        bail!("refusing to write outside the FTL data directory");
2788    }
2789
2790    // Prefer the tag matching this binary; fall back to master.
2791    let version = env!("CARGO_PKG_VERSION");
2792    let refs = [format!("v{version}"), "master".to_string()];
2793    let client = reqwest::Client::new();
2794    let mut fetched = 0u32;
2795
2796    for (name, path_tmpl, out_name) in selected {
2797        let repo_path = path_tmpl.replace("{locale}", &locale);
2798        let mut body: Option<String> = None;
2799        for git_ref in &refs {
2800            let url = format!(
2801                "https://raw.githubusercontent.com/zeroclaw-labs/zeroclaw/{git_ref}/{repo_path}"
2802            );
2803            let resp = client.get(&url).send().await?;
2804            if resp.status().is_success() {
2805                body = Some(resp.text().await?);
2806                break;
2807            }
2808        }
2809        match body {
2810            Some(content) => {
2811                let out_path = dest.join(out_name);
2812                std::fs::write(&out_path, content)
2813                    .with_context(|| format!("writing {}", out_path.display()))?;
2814                println!(
2815                    "{}",
2816                    ta(
2817                        "cli-locales-fetched",
2818                        &[("name", name), ("path", &out_path.display().to_string())],
2819                        "fetched catalogue",
2820                    )
2821                );
2822                fetched += 1;
2823            }
2824            None => {
2825                eprintln!(
2826                    "{}",
2827                    ta(
2828                        "cli-locales-skipped",
2829                        &[
2830                            ("name", name),
2831                            ("path", &repo_path),
2832                            ("refs", &refs.join(", "))
2833                        ],
2834                        "skipped: not on upstream",
2835                    )
2836                );
2837            }
2838        }
2839    }
2840
2841    if fetched == 0 {
2842        bail!("no catalogues fetched for locale '{locale}'");
2843    }
2844    println!(
2845        "{}",
2846        ta(
2847            "cli-locales-installed",
2848            &[
2849                ("count", &fetched.to_string()),
2850                ("locale", &locale),
2851                ("dir", &dest.display().to_string())
2852            ],
2853            "Installed catalogues",
2854        )
2855    );
2856    Ok(())
2857}

5698/// What the `get-paircode` CLI command should do against the running gateway.

5703#[cfg(feature = "agent-runtime")]
5704enum PaircodeAction {
5705    /// GET the current code; do not mint or revoke anything.
5706    Show,
5707    /// Issue a fresh code for an additional client; revoke nothing.
5708    AddClient,
5709    /// Revoke every paired token + clear the registry, then issue a code.
5710    RotateAll,

5713}

5725            self,
5726            PaircodeAction::RotateAll | PaircodeAction::RotateDevice(_)

5733            PaircodeAction::RotateAll => Some("all".to_string()),

5761    port: u16,
5762    path_prefix: Option<&str>,
5763    action: &PaircodeAction,
5764) -> Result<PaircodeResult> {
5765    let client = reqwest::Client::new();
5766
5767    let response = if action.mints_code() {
5768        let mut url = gateway_admin_url(host, port, path_prefix, "/admin/paircode/new");
5769        if let Some(rotate) = action.rotate_query() {
5770            url.push_str("?rotate=");
5771            url.push_str(&urlencoding::encode(&rotate));
5772        }
5773        client
5774            .post(&url)
5775            .timeout(std::time::Duration::from_secs(5))
5776            .send()
5777            .await
5778    } else {
5779        let url = gateway_admin_url(host, port, path_prefix, "/admin/paircode");
5780        client
5781            .get(&url)
5782            .timeout(std::time::Duration::from_secs(5))
5783            .send()
5784            .await
5785    };
5786
5787    let response = response.map_err(|e| {
5788        ::zeroclaw_log::record!(
5789            WARN,
5790            ::zeroclaw_log::Event::new(module_path!(), ::zeroclaw_log::Action::Fail)
5791                .with_outcome(::zeroclaw_log::EventOutcome::Failure)
5792                .with_attrs(::serde_json::json!({"error": format!("{}", e)})),
5793            "gateway paircode fetch: connect failed"
5794        );
5795        anyhow::Error::msg(format!("Failed to connect to gateway: {e}"))
5796    })?;
5797
5798    // A rotation that revoked tokens but could not issue a code (registry
5799    // disabled, device not found, persist failure) returns a non-2xx with an
5800    // explanatory message in the body. Surface that message rather than
5801    // collapsing it into a bare status line, so the operator knows what
5802    // state the gateway is in after the revoke attempt.
5803    let status = response.status();
5804    let json: serde_json::Value = response.json().await.map_err(|e| {
5805        ::zeroclaw_log::record!(
5806            WARN,
5807            ::zeroclaw_log::Event::new(module_path!(), ::zeroclaw_log::Action::Fail)
5808                .with_outcome(::zeroclaw_log::EventOutcome::Failure)
5809                .with_attrs(
5810                    ::serde_json::json!({"error": format!("{}", e), "status": status.as_u16()})
5811                ),
5812            "gateway paircode response: JSON parse failed"
5813        );
5814        anyhow::Error::msg(format!("Gateway responded with status {status}: {e}"))
5815    })?;
5816
5817    let message = json
5818        .get("message")
5819        .and_then(|v| v.as_str())
5820        .map(String::from);
5821
5822    if json.get("success").and_then(|v| v.as_bool()) != Some(true) {
5823        if !status.is_success() {
5824            ::zeroclaw_log::record!(
5825                WARN,
5826                ::zeroclaw_log::Event::new(module_path!(), ::zeroclaw_log::Action::Fail)
5827                    .with_outcome(::zeroclaw_log::EventOutcome::Failure)
5828                    .with_attrs(::serde_json::json!({"status": status.as_u16()})),
5829                "gateway paircode fetch returned non-success status"
5830            );
5831        }
5832        return Ok(PaircodeResult::NoCode { message });
5833    }
5834
5835    match json.get("pairing_code").and_then(|v| v.as_str()) {
5836        Some(code) => Ok(PaircodeResult::Code {
5837            code: code.to_string(),
5838            message,
5839        }),
5840        None => Ok(PaircodeResult::NoCode { message }),
5841    }
5842}

6151/// Gate every serving entry point on the loaded security posture.
6152///
6153/// `Config.degraded_security` lists security-critical sections (`security`,
6154/// `risk_profiles`, `peer_groups`) that failed to parse and were reset to
6155/// their defaults during load — the running posture may then be WEAKER than
6156/// intended. "Secure by default" must fail loud here (FND-006 §4.5), so every
6157/// path that brings up the serving surface (daemon, `gateway start`,
6158/// `gateway restart`) routes through this before binding.
6159///
6160/// Returns `Err` when degraded and `allow_degraded` is false (the caller
6161/// propagates and the process never serves). When degraded and explicitly
6162/// allowed, boots but returns the `JoinHandle` of a repeating-WARN nag task so
6163/// the caller can abort it on reload; `Ok(None)` means a clean posture.
6164fn gate_security_posture(
6165    config: &zeroclaw::config::Config,
6166    allow_degraded: bool,
6167) -> anyhow::Result<Option<tokio::task::JoinHandle<()>>> {
6168    if config.degraded_security.is_empty() {
6169        return Ok(None);
6170    }
6171    let sections = config.degraded_security.join(", ");
6172    if !allow_degraded {
6173        anyhow::bail!(
6174            "Config contains malformed security-critical sections ({sections}); \
6175             they were reset to defaults, so the running posture may be weaker \
6176             than intended. Refusing to serve with a degraded security posture. \
6177             Repair these sections in {} and restart — run `zeroclaw config \
6178             migrate` to see the precise error. To boot anyway (e.g. to reach \
6179             the gateway config editor and repair from there), re-run with \
6180             `--allow-degraded-security`.",
6181            config.config_path.display()
6182        );
6183    }
6184    let config_path = config.config_path.display().to_string();
6185    let handle = ::zeroclaw_spawn::spawn!(async move {
6186        let mut ticker = tokio::time::interval(std::time::Duration::from_secs(30));
6187        loop {
6188            ticker.tick().await;
6189            ::zeroclaw_log::record!(
6190                WARN,
6191                ::zeroclaw_log::Event::new(module_path!(), ::zeroclaw_log::Action::Note)
6192                    .with_outcome(::zeroclaw_log::EventOutcome::Unknown)
6193                    .with_attrs(::serde_json::json!({ "degraded_security": sections })),
6194                &format!(
6195                    "Running with DEGRADED security: sections ({sections}) were reset to \
6196                     defaults and `--allow-degraded-security` was set. The posture may be \
6197                     weaker than intended — repair {config_path} and reload \
6198                     (SIGUSR1 / `zeroclaw admin reload`) as soon as possible."
6199                )
6200            );
6201        }
6202    });
6203    Ok(Some(handle))
6204}
