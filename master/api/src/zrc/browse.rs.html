1//! `zeroclaw browse [path]` — CLI adapter over
2//! `zeroclaw_runtime::browse::list_directory`. Thin print formatter; the
3//! walking + containment rule lives in the runtime crate so the gateway
4//! and the CLI share one implementation.
5
6use anyhow::Result;
7use zeroclaw_runtime::browse::list_directory;
8use zeroclaw_runtime::i18n::{get_required_cli_string, get_required_cli_string_with_args};
9
10pub fn handle_browse(path: String, config: &crate::config::Config) -> Result<()> {
11    let result = list_directory(config, &path)?;
12    let display_path = if result.path.is_empty() {
13        "/"
14    } else {
15        &result.path
16    };
17    println!(
18        "{}",
19        get_required_cli_string_with_args(
20            "cli-browse-header",
21            &[
22                (
23                    "path",
24                    &console::style(format!("shared/{display_path}"))
25                        .white()
26                        .bold()
27                        .to_string(),
28                ),
29                ("count", &result.entries.len().to_string()),
30            ],
31        )
32    );
33    if result.entries.is_empty() {
34        println!("  {}", get_required_cli_string("cli-browse-empty"));
35        return Ok(());
36    }
37    for entry in result.entries {
38        match entry.kind {
39            "dir" => println!("  {}/", console::style(&entry.name).cyan().bold()),
40            _ => match entry.size {
41                Some(s) => println!(
42                    "  {}",
43                    get_required_cli_string_with_args(
44                        "cli-browse-file-bytes",
45                        &[
46                            ("name", &console::style(&entry.name).dim().to_string()),
47                            ("bytes", &s.to_string()),
48                        ],
49                    )
50                ),
51                None => println!("  {}", console::style(&entry.name).dim()),
52            },
53        }
54    }
55    Ok(())
56}
